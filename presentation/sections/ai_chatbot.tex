\section{Feature 3: Personalized AI Assistant}

\begin{frame}{Context-Aware Conversational AI}
    \begin{itemize}
        \item \textbf{Multi-Context Integration}: Chatbot retrieves data from three sources:
            \begin{enumerate}
                \item \textbf{User Profile}: (Conditions, Allergies, BMI).
                \item \textbf{Product Context}: (Nutrition facts, Ingredients).
                \item \textbf{Analysis Context}: (Verdict reasoning, confidence score).
            \end{enumerate}
        \item \textbf{Safety First}: Hard-coded disclaimers ensuring responses are informational only.
        \item \textbf{Hybrid Architecture}: Deterministic fallback for critical queries when AI service is unavailable.
    \end{itemize}
\end{frame}
